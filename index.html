Advanced Linux Snippets

$ find /var/log -name "*.log" -mtime -7 -exec grep -l "ERROR" {} +

$ strace -c -f -p $(pgrep nginx) 2>&1 | tail -20

$ rsync -avz --delete --info=progress2 --partial --bwlimit=8000 /data/ user@backup:/archive/
